                bandB: { ldmMin: -1, ldmMax: 1, ldmStep: 0.2, ldmCap: 6.3, wtMin: -0.3, wtMax: 0.2, wtStep: 0.2, wtCap: 3.2 },

                bandOther: { ldmMin: -0.6, ldmMax: 0.6, ldmStep: 0.2, ldmCap: 0, wtMin: -0.4, wtMax: 0.4, wtStep: 0.2, wtCap: 0 }

    // Shallow-aware recursive merge: lets old saved blobs pick up new default
    // keys (e.g. after a script update adds a field) without losing user edits.
    function deepMergeDefaults(base, override) {
        let result = Array.isArray(base) ? base.slice() : Object.assign({}, base);
        if (!override || typeof override !== 'object') return result;
        for (let key in override) {
            let baseVal = base ? base[key] : undefined;
            let overVal = override[key];
            if (overVal && typeof overVal === 'object' && !Array.isArray(overVal) && baseVal && typeof baseVal === 'object') {
                result[key] = deepMergeDefaults(baseVal, overVal);
            } else if (overVal !== undefined) {
                result[key] = overVal;
            }
        }
        return result;
    }

    // v18.0/18.1 stored tag inclusion as boolean maps keyed by the (then fixed)
    // tag text. v18.2 made the tag lists themselves editable, so they're now
    // ordered arrays. This carries forward anyone's existing checkbox choices
    // the first time their saved config is loaded under the new format.
    function migrateLegacyConfig(saved, defaults) {
        if (saved && saved.bodyTypes && !saved.bodyTypeTags) {
            defaults.bodyTypeTags = BODY_TYPES.map(t => ({ text: t, enabled: saved.bodyTypes[t] !== false }));
            delete saved.bodyTypes;
        }
        if (saved && saved.vehicleSizes && !saved.vehicleSizeTags) {
            defaults.vehicleSizeTags = VEHICLE_SIZES.map(t => ({ text: t, enabled: saved.vehicleSizes[t] !== false }));
            delete saved.vehicleSizes;
        }
        // v18.4 had a single "enabled" flag covering loading only; v18.5 split
        // it into obfuscateLoading/obfuscateDelivery. Carry the old value over.
        if (saved && saved.locationObfuscation && 'enabled' in saved.locationObfuscation && !('obfuscateLoading' in saved.locationObfuscation)) {
            defaults.locationObfuscation.obfuscateLoading = saved.locationObfuscation.enabled;
            delete saved.locationObfuscation.enabled;
        }
        return defaults;
    }

    function loadPostingConfig() {
        let defaults = defaultPostingConfig();
        try {
            let raw = localStorage.getItem('tm-cfg-posting');
            if (!raw) return defaults;
            let saved = JSON.parse(raw);
            defaults = migrateLegacyConfig(saved, defaults);
            return deepMergeDefaults(defaults, saved);
        } catch (e) {
            return defaults;
        }
    }

    function savePostingConfig() {
        localStorage.setItem('tm-cfg-posting', JSON.stringify(config.posting));
    }

    config.posting = loadPostingConfig();

    // --- Core Extraction Utilities (Reads from Main Page) ---
    function getTextByTestId(id) {
        const el = document.querySelector(`[data-testid="${id}"]`);
        return el ? el.textContent.replace(/\u00A0/g, ' ').trim() : "";
    }

    function extractDigits(text) {
        if (!text) return "";
        const match = text.match(/[\d.]+/);
        return match ? match[0] : "";
    }

    function mergeRange(earliest, latest, defaultVal) {
        if (!earliest && !latest) return defaultVal;
        let cleanE = earliest ? earliest.replace(/\.$/, '').trim() : "";
        let cleanL = latest ? latest.replace(/\.$/, '').trim() : "";

        if (cleanE && cleanL && cleanE !== cleanL) {
            if (cleanE.includes('-')) return cleanE;
            return `${cleanE} - ${cleanL}`;
        }
        return cleanE || cleanL;
    }

    function offsetTimeStr(timeStr, hoursToAdd) {
        if (!timeStr || !timeStr.includes(':')) return timeStr;
        if (hoursToAdd === 0 || isNaN(hoursToAdd)) return timeStr;

        let parts = timeStr.split(':');
        let h = parseInt(parts[0], 10);
        let m = parseInt(parts[1], 10);
        if (isNaN(h) || isNaN(m)) return timeStr;

        h += hoursToAdd;
        if (h >= 24) h = h % 24;
        if (h < 0) h = 24 + (h % 24);

        let hStr = h < 10 ? '0' + h : h;
        let mStr = m < 10 ? '0' + m : m;
        return `${hStr}:${mStr}`;
    }

    // --- Asynchronous TinyURL Shortener ---
    function fetchShortenedUrl(longUrl) {
        return new Promise((resolve) => {
            GM.xmlHttpRequest({
                method: 'GET',
                url: `https://tinyurl.com/api-create.php?url=${encodeURIComponent(longUrl)}`,
                timeout: 4000,
                onload: function(response) {
                    if (response.status === 200 && response.responseText) {
                        resolve(response.responseText.trim());
                    } else {
                        resolve(longUrl);
                    }
                },
                onerror: function() { resolve(longUrl); },
                ontimeout: function() { resolve(longUrl); }
            });
        });
    }

    // --- Multi-Stop Collector ---
    // Walks LoadingPlacesView/*_N for N = 0,1,2,... until no place is found.
    // Each TimoCom "loading place" row is a stop (loading OR unloading); a round
    // trip A -> B -> B -> A is exposed as four stops: A, B, B, A.
    function collectStops() {
        let stops = [];
        for (let i = 0; i < 12; i++) {
            let country = getTextByTestId(`LoadingPlacesView/loadingPlaceCountry_${i}`);
            let city = getTextByTestId(`LoadingPlacesView/loadingPlaceCity_${i}`);
            if (!country && !city) break;

            let eDate = getTextByTestId(`LoadingPlacesView/earliestLoadingDate_${i}`);
            let lDate = getTextByTestId(`LoadingPlacesView/latestLoadingDate_${i}`);
            let eTime = getTextByTestId(`LoadingPlacesView/earliestLoadingTime_${i}`);
            let lTime = getTextByTestId(`LoadingPlacesView/latestLoadingTime_${i}`);

            stops.push({
                index: i,
                country: country,
                city: city,
                addr: `${country} ${city}`.trim(),
                eDate: eDate,
                lDate: lDate,
                eTime: eTime,
                lTime: lTime
            });
        }
        return stops;
    }

    // --- Google Maps URL Builder (consecutive-duplicate aware) ---
    // Keeps every address in order, but collapses any address that is listed
    // immediately after an identical one. So [A, B, B, A] -> [A, B, A]
    // (B appears once), while a non-round trip [A, B, C] is left untouched.
    function buildMapsUrl(stops) {
        let addrs = stops.map(s => s.addr).filter(a => a);

        let deduped = [];
        for (let a of addrs) {
            let prev = deduped.length ? deduped[deduped.length - 1] : null;
            if (!prev || prev.toLowerCase() !== a.toLowerCase()) deduped.push(a);
        }

        if (deduped.length === 0) return "";
        if (deduped.length === 1) {
            return `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(deduped[0])}`;
        }

        let origin = deduped[0];
        let destination = deduped[deduped.length - 1];
        let waypoints = deduped.slice(1, -1);

        let url = `https://www.google.com/maps/dir/?api=1&origin=${encodeURIComponent(origin)}&destination=${encodeURIComponent(destination)}&travelmode=driving`;
        if (waypoints.length) {
            url += `&waypoints=${waypoints.map(w => encodeURIComponent(w)).join('|')}`;
        }
        return url;
    }

    // --- Postcode Extraction ---
    // City field looks like ", 41001 Sevilla" / ", 1234 AB Amsterdam".
    // Pulls the leading postcode token(s): numeric tokens, plus a trailing
    // short all-caps token (NL/UK second segment). Falls back to the city name.
    // City field looks like ", 41001 Sevilla" / ", 1234 AB Amsterdam" / just
    // ", Westertimke" when there's no postcode at all. Splits it into the
    // leading postcode-like tokens (digits, plus a trailing short all-caps
    // suffix like NL/UK use) and whatever city name text follows. When no
    // postcode tokens are found, the whole string is treated as the city
    // name (matching extractPostcode's long-standing fallback behavior).
    function splitPostcodeAndCity(city) {
        if (!city) return { postcode: "", cityName: "" };
        let c = city.replace(/^[\s,]+/, '').trim();
        if (!c) return { postcode: "", cityName: "" };
        let tokens = c.split(/\s+/);
        let pc = [];
        let i = 0;
        for (; i < tokens.length; i++) {
            let t = tokens[i];
            if (/\d/.test(t)) { pc.push(t); continue; }
            if (pc.length && /^[A-Z]{1,3}$/.test(t)) { pc.push(t); continue; }
            break;
        }
        if (pc.length) return { postcode: pc.join(' '), cityName: tokens.slice(i).join(' ') };
        return { postcode: "", cityName: c };
    }

    function extractPostcode(city) {
        let parts = splitPostcodeAndCity(city);
        return parts.postcode || parts.cityName;
    }

    // --- Posting Export: "Loading address" / "Delivery address" Lines ---
    // "[Country Code] [postcode] [city]" — skips any piece that's missing
    // (e.g. no postcode found) rather than leaving a stray double space.
    function buildAddressLine(stop) {
        if (!stop) return '';
        let { postcode, cityName } = splitPostcodeAndCity(stop.city);
        return [stop.country, postcode, cityName].filter(p => p && p.trim()).join(' ');
    }

    function buildAddressLineFromPlace(place) {
        if (!place) return '';
        return [place.countryCode, place.postcode, place.city].filter(p => p && p.trim()).join(' ');
    }

    // --- Country-Code + Postcode Array Builder (shared) ---
    // Produces the ordered, consecutive-duplicate-collapsed "CC-postcode" list
    // that both the "All" and "First+Last" route modes are built from, and
    // that the Posting export's loading-location obfuscation also edits.
    function getCodesArray(stops, separator) {
        separator = separator || '-';
        let parts = stops
            .map(s => {
                let pc = extractPostcode(s.city);
                return s.country ? (pc ? `${s.country}${separator}${pc}` : s.country) : pc;
            })
            .filter(p => p);

        let deduped = [];
        for (let p of parts) {
            let prev = deduped.length ? deduped[deduped.length - 1] : null;
            if (!prev || prev.toLowerCase() !== p.toLowerCase()) deduped.push(p);
        }
        return deduped;
    }

    // --- Country-Code + Postcode Route Builder ---
    // Same consecutive-duplicate collapsing as the map URL, rendered as
    // "ES-41001 > HU-5630" (every loading/destination stop included).
    function buildCodesRoute(stops) {
        return getCodesArray(stops).join(' > ');
    }

    // --- Posting Export: LDM / Weight Randomization ---
    // Builds the list of valid offsets between min/max stepped by `step`,
    // anchored at `min` (so ranges that don't divide evenly, e.g. -0.3..0.2
    // in 0.2 steps, still produce a well-defined, reproducible candidate set).
    function genStepCandidates(min, max, step) {
        let vals = [];
        step = Math.abs(step) || 0.2;
        let v = min;
        let guard = 0;
        while (v <= max + 1e-9 && guard < 500) {
            vals.push(Math.round(v * 100) / 100);
            v += step;
            guard++;
        }
        if (!vals.length) vals.push(0);
        return vals;
    }

    function randomOffset(min, max, step) {
        let candidates = genStepCandidates(min, max, step);
        return candidates[Math.floor(Math.random() * candidates.length)];
    }

    // Applies the configured jitter rules to a raw LDM/weight pair. Always
    // picks a band — Band A, Band B, or the "Other" fallback for anything
    // outside both — so every call (i.e. every click of "Organise & Copy for
    // Posting") produces a freshly randomized result, never a pass-through.
    // A small safety floor (0.2 LDM / 0.1t) keeps the offset from ever
    // producing a zero or negative value. A cap of 0 means "no hard ceiling".
    function jitterDims(ldmRaw, wtRaw) {
        let ldm = parseFloat(ldmRaw);
        let wt = parseFloat(wtRaw);
        if (isNaN(ldm)) ldm = 0;
        if (isNaN(wt)) wt = 0;

        let j = config.posting.jitter;
        let band;
        if (ldm < 4 && wt < 1) {
            band = j.bandA;
        } else if (ldm >= 4 && ldm <= 6.3 && wt >= 1 && wt <= 3.2) {
            band = j.bandB;
        } else {
            band = j.bandOther;
        }

        ldm += randomOffset(band.ldmMin, band.ldmMax, band.ldmStep);
        wt += randomOffset(band.wtMin, band.wtMax, band.wtStep);

        ldm = Math.max(0.2, ldm);
        wt = Math.max(0.1, wt);
        if (band.ldmCap > 0) ldm = Math.min(band.ldmCap, ldm);
        if (band.wtCap > 0) wt = Math.min(band.wtCap, wt);

        return {
            ldm: Math.round(ldm * 10) / 10,
            wt: Math.round(wt * 10) / 10
        };
    }

    // --- Posting Export: Date Format ("22.06" -> "22/06") ---
    function slashDate(str) {
        return str ? str.replace(/\./g, '/') : str;
    }

    // --- Posting Export: Configurable Line Labels ---
    // Lets the field name in front of each line (e.g. "Wielkość pojazdu") be
    // renamed from the settings panel (e.g. to "vehicleSizeId"). An empty
    // label drops the "Label: " prefix entirely instead of leaving a bare colon.
    function withLabel(label, value) {
        return label ? `${label}: ${value}` : value;
    }

    // --- Posting Export: Keyword-Only Notes ---
    // Distills the free-text description/remarks down to a short, normalized
    // keyword list (e.g. "ADR, LIFT") rather than re-posting the raw text.
    function extractNotesKeywords(text) {
        if (!text) return "";
        let found = [];
        if (/\badr/i.test(text)) found.push('ADR');
        if (/\blift/i.test(text)) found.push('LIFT');
        if (/\b(pallet|plt|pal|epal)/i.test(text)) found.push('PALLETS');
        return found.join(', ');
    }

    // --- Posting Export: "Paste Freight Info" Parser ---
    // Pulls loading/delivery country+postcode, dates/times, LDM/weight, and
    // notes text out of arbitrary pasted freight info — built to read this
    // script's own export formats (old hyphenated or new space-separated
    // codes, "Loading:"/"Delivery:"/"LDM/ Wt:" lines) but tolerant of minor
    // variations. Returns a stops[] array shaped exactly like collectStops(),
    // so every downstream function (route building, obfuscation, jitter,
    // labels, tags) runs completely unchanged regardless of the data source.
    function parseDateTimeFragment(str) {
        if (!str) return { date: '', time: '' };
        let timeMatch = str.match(/(\d{1,2}):(\d{2})/);
        let time = timeMatch ? `${timeMatch[1].padStart(2, '0')}:${timeMatch[2]}` : '';
        let rangeMatch = str.match(/(\d{1,2}[.\/]\d{1,2}(?:[.\/]\d{2,4})?)\s*-\s*(\d{1,2}[.\/]\d{1,2}(?:[.\/]\d{2,4})?)/);
        if (rangeMatch) return { date: `${rangeMatch[1].replace(/\//g, '.')} - ${rangeMatch[2].replace(/\//g, '.')}`, time };
        let singleMatch = str.match(/\d{1,2}[.\/]\d{1,2}(?:[.\/]\d{2,4})?/);
        return { date: singleMatch ? singleMatch[0].replace(/\//g, '.') : '', time };
    }

    function parsePastedFreightText(text) {
        if (!text || !text.trim()) return null;

        // "NL-7511", "DE 26452", "ES14009", optionally with a trailing
        // letter segment (e.g. NL postcodes' "AB" suffix).
        let codeMatches = [...text.matchAll(/\b([A-Z]{2})[\s-]+(\d{3,6}(?:[\s-][A-Z]{1,3})?)\b/g)];

        let loadingCountry = '', loadingPc = '', deliveryCountry = '', deliveryPc = '';
        if (codeMatches.length) {
            loadingCountry = codeMatches[0][1];
            loadingPc = codeMatches[0][2].trim();
            let last = codeMatches[codeMatches.length - 1];
            deliveryCountry = last[1];
            deliveryPc = last[2].trim();
        }

        let loadFrag = (text.match(/loading[^:]*:\s*([^\n\r]+)/i) || [])[1] || '';
        let delFrag = (text.match(/delivery[^:]*:\s*([^\n\r]+)/i) || [])[1] || '';
        let loadDT = parseDateTimeFragment(loadFrag);
        let delDT = parseDateTimeFragment(delFrag);

        // Prefer a dedicated LDM/weight line if one is found; fall back to
        // scanning the whole text for a "<number>m" / "<number>t" pattern.
        let dimsLine = (text.match(/(?:ldm\s*\/?\s*wt|ldm|weight)[^:]*:\s*([^\n\r]+)/i) || [])[1] || text;
        let ldmMatch = dimsLine.match(/(\d+(?:[.,]\d+)?)\s*m\b/i) || text.match(/(\d+(?:[.,]\d+)?)\s*m\b/i);
        let wtMatch = dimsLine.match(/(\d+(?:[.,]\d+)?)\s*t\b/i) || text.match(/(\d+(?:[.,]\d+)?)\s*t\b/i);
        let ldmRaw = ldmMatch ? ldmMatch[1].replace(',', '.') : '';
        let wtRaw = wtMatch ? wtMatch[1].replace(',', '.') : '';

        let stops = [];
        if (loadingCountry || loadDT.date) {
            stops.push({
                index: 0, country: loadingCountry, city: loadingPc ? `, ${loadingPc}` : '',
                addr: `${loadingCountry} ${loadingPc}`.trim(), eDate: '', lDate: loadDT.date, eTime: '', lTime: loadDT.time
            });
        }
        if (deliveryCountry || delDT.date) {
            stops.push({
                index: 1, country: deliveryCountry, city: deliveryPc ? `, ${deliveryPc}` : '',
                addr: `${deliveryCountry} ${deliveryPc}`.trim(), eDate: '', lDate: delDT.date, eTime: '', lTime: delDT.time
            });
        }

        // Nothing at all recognized — let the caller show an error instead of
        // silently producing an empty/placeholder export.
        if (!stops.length && !ldmRaw && !wtRaw) return null;

        return { stops, ldmRaw, wtRaw, notesText: text };
    }

    // --- Posting Export: Loading / Delivery Location Obfuscation ---
    // Replaces the LOADING and/or DELIVERY stop's "CC-postcode" with a real
    // nearby place at a random distance inside [minKm, maxKm] from the true
    // postcode, in a random direction. Each end is toggled independently;
    // any other (intermediate) stops are always left untouched.
    // Two free, no-key public APIs are used (GM.xmlHttpRequest bypasses CORS):
    //   1. api.zippopotam.us   — forward geocode: postcode -> lat/lon
    //   2. nominatim.openstreetmap.org — reverse geocode: lat/lon -> real place
    // Any failure (unsupported country, network error, no result) falls back
    // to leaving the real code in place rather than breaking the copy.

    function deg2rad(d) { return d * Math.PI / 180; }

    // Strips a postcode down to its digit/hyphen core for the geocoding APIs
    // (e.g. NL "7511 AB" -> "7511"; PL "30-001" is already fine as-is).
    function postcodeForGeocode(postcode) {
        if (!postcode) return "";
        let trimmed = postcode.trim();
        if (!trimmed) return "";

        // Digit-led postcodes (NL "7511 AB", PL "30-001", DE/FR/ES 5-digit,
        // etc.) — Zippopotam wants just the leading digit/hyphen core,
        // without any trailing letter suffix.
        if (/^\d/.test(trimmed)) {
            let m = trimmed.match(/\d[\d\s-]*\d|\d+/);
            return m ? m[0].replace(/\s+/g, '') : trimmed;
        }

        // Letter-led / alphanumeric postcodes (UK "PE19 2", Canada
        // "K1A 0B1") — the letters ARE the postcode; stripping them to
        // digits-only (e.g. "PE19 2" -> "192") destroys the part that
        // actually identifies the location. Just normalize whitespace.
        return trimmed.replace(/\s+/g, ' ');
    }

    function geocodePostcode(countryCode, postcode) {
        return new Promise((resolve) => {
            if (!countryCode || !postcode) { resolve(null); return; }
            let url = `https://api.zippopotam.us/${encodeURIComponent(countryCode.trim().toLowerCase())}/${encodeURIComponent(postcode)}`;
            console.log('[TM-Obfuscate] GET', url);
            GM.xmlHttpRequest({
                method: 'GET',
                url,
                timeout: 4000,
                onload: (resp) => {
                    try {
                        console.log('[TM-Obfuscate] zippopotam response status=%s body=%s', resp.status, (resp.responseText || '').slice(0, 300));
                        if (resp.status !== 200) { resolve(null); return; }
                        let data = JSON.parse(resp.responseText);
                        let place = data.places && data.places[0];
                        if (!place) { resolve(null); return; }
                        let lat = parseFloat(place.latitude);
                        let lon = parseFloat(place.longitude);
                        if (isNaN(lat) || isNaN(lon)) { console.warn('[TM-Obfuscate] zippopotam returned a place with non-numeric lat/lon:', place); resolve(null); return; }
                        resolve({ lat, lon });
                    } catch (e) { console.error('[TM-Obfuscate] zippopotam response parse error:', e); resolve(null); }
                },
                onerror: (err) => { console.error('[TM-Obfuscate] zippopotam request ERROR (likely blocked/CORS/permission):', err); resolve(null); },
                ontimeout: () => { console.warn('[TM-Obfuscate] zippopotam request TIMED OUT'); resolve(null); }
            });
        });
    }

    // --- Nominatim Rate Limiter ---
    // Nominatim's usage policy caps public requests at ~1/second. Loading and
    // delivery obfuscation fire their reverse-geocode calls concurrently (and
    // repeated clicks can come in quick succession), which was silently
    // hitting that limit and falling back to the real address far more often
    // than it should. This serializes every reverse-geocode call through a
    // shared queue with a minimum spacing, regardless of how many lookups are
    // in flight at once or how fast the buttons get clicked.
    let nominatimQueueReadyAt = 0;
    const NOMINATIM_MIN_INTERVAL_MS = 1100;

    function waitForNominatimSlot() {
        return new Promise((resolve) => {
            let now = Date.now();
            let runAt = Math.max(now, nominatimQueueReadyAt);
            nominatimQueueReadyAt = runAt + NOMINATIM_MIN_INTERVAL_MS;
            setTimeout(resolve, runAt - now);
        });
    }

    function reverseGeocodeOnce(lat, lon, zoom) {
        return new Promise((resolve) => {
            let url = `https://nominatim.openstreetmap.org/reverse?format=jsonv2&lat=${lat}&lon=${lon}&zoom=${zoom || 14}&addressdetails=1`;
            console.log('[TM-Obfuscate] GET', url);
            GM.xmlHttpRequest({
                method: 'GET',
                url,
                timeout: 5000,
                headers: { 'Accept-Language': 'en', 'User-Agent': 'TimoCom-Cargo-Formatter-UserScript' },
                onload: (resp) => {
                    try {
                        console.log('[TM-Obfuscate] nominatim reverse response status=%s body=%s', resp.status, (resp.responseText || '').slice(0, 300));
                        if (resp.status !== 200) { resolve(null); return; }
                        let data = JSON.parse(resp.responseText);
                        let addr = data.address || {};
                        let cc = (addr.country_code || '').toUpperCase();
                        if (!cc) { resolve(null); return; }
                        resolve({ countryCode: cc, postcode: addr.postcode || '', city: addr.city || addr.town || addr.village || addr.municipality || '' });
                    } catch (e) { console.error('[TM-Obfuscate] nominatim reverse parse error:', e); resolve(null); }
                },
                onerror: (err) => { console.error('[TM-Obfuscate] nominatim reverse request ERROR (likely blocked/CORS/permission):', err); resolve(null); },
                ontimeout: () => { console.warn('[TM-Obfuscate] nominatim reverse request TIMED OUT'); resolve(null); }
            });
        });
    }

    // Serialized through the rate limiter above, with one retry (after
    // re-queuing for another slot) if the first attempt comes back empty —
    // covers transient throttling/network blips rather than giving up
    // immediately and silently falling back to the real address.
    async function reverseGeocode(lat, lon, zoom) {
        await waitForNominatimSlot();
        let result = await reverseGeocodeOnce(lat, lon, zoom);
        if (result) return result;

        await waitForNominatimSlot();
        return reverseGeocodeOnce(lat, lon, zoom);
    }

    // --- Forward Geocode By City Name (Nominatim) ---
    // Zippopotam (geocodePostcode above) only understands actual postal
    // codes. Plenty of real addresses have none — e.g. a bare city name with
    // no digits at all — which used to send that city name to Zippopotam AS
    // IF it were a postcode, fail every time, and silently fall back to the
    // real address regardless of anything else. This is the missing other
    // half: geocode by name instead, through the same Nominatim instance
    // (so it goes through the same rate-limit queue) when there's no
    // postcode to work with.
    function geocodeCityNameOnce(countryCode, cityName) {
        return new Promise((resolve) => {
            if (!countryCode || !cityName) { resolve(null); return; }
            let url = `https://nominatim.openstreetmap.org/search?format=jsonv2&city=${encodeURIComponent(cityName)}&countrycodes=${encodeURIComponent(countryCode.trim().toLowerCase())}&limit=1`;
            console.log('[TM-Obfuscate] GET', url);
            GM.xmlHttpRequest({
                method: 'GET',
                url,
                timeout: 5000,
                headers: { 'Accept-Language': 'en', 'User-Agent': 'TimoCom-Cargo-Formatter-UserScript' },
                onload: (resp) => {
                    try {
                        console.log('[TM-Obfuscate] nominatim search response status=%s body=%s', resp.status, (resp.responseText || '').slice(0, 300));
                        if (resp.status !== 200) { resolve(null); return; }
                        let data = JSON.parse(resp.responseText);
                        if (!data || !data.length) { resolve(null); return; }
                        let lat = parseFloat(data[0].lat);
                        let lon = parseFloat(data[0].lon);
                        if (isNaN(lat) || isNaN(lon)) { resolve(null); return; }
                        resolve({ lat, lon });
                    } catch (e) { console.error('[TM-Obfuscate] nominatim search parse error:', e); resolve(null); }
                },
                onerror: (err) => { console.error('[TM-Obfuscate] nominatim search request ERROR (likely blocked/CORS/permission):', err); resolve(null); },
                ontimeout: () => { console.warn('[TM-Obfuscate] nominatim search request TIMED OUT'); resolve(null); }
            });
        });
    }

    async function geocodeCityName(countryCode, cityName) {
        await waitForNominatimSlot();
        let result = await geocodeCityNameOnce(countryCode, cityName);
        if (result) return result;

        await waitForNominatimSlot();
        return geocodeCityNameOnce(countryCode, cityName);
    }

    // Generic "find a real nearby place" lookup for any single stop — used
    // for both the loading and delivery obfuscation toggles below.

    // Great-circle distance between two points, km — used to score candidate
    // points against the history of previously-picked points for this offer.
    function haversineKm(lat1, lon1, lat2, lon2) {
        const R = 6371;
        const dLat = deg2rad(lat2 - lat1);
        const dLon = deg2rad(lon2 - lon1);
        const a = Math.sin(dLat / 2) ** 2 + Math.cos(deg2rad(lat1)) * Math.cos(deg2rad(lat2)) * Math.sin(dLon / 2) ** 2;
        return R * 2 * Math.asin(Math.sqrt(Math.min(1, a)));
    }

    // --- Overpass API: Real Cities/Towns/Villages By Population ---
    // Queries OpenStreetMap (via the free, no-key Overpass API) for populated
    // places within maxKm of the origin, then filters to the [minKm, maxKm]
    // annulus and sorts descending by population. This is what makes "largest
    // city first, then next-largest, etc." possible — picking a random point
    // and reverse-geocoding it (the old approach) had no notion of city size.
    // Trimmed, lowercased, diacritic-stripped — for exact (not substring)
    // name comparisons, so accent/encoding differences between Overpass and
    // the stop's own city text don't cause a false mismatch.
    function normalizeNameForCompare(s) {
        return (s || '').trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
    }

    function cleanCityNameForSearch(name) {
        if (!name) return "";
        return name.replace(/\s*\([^)]*\)\s*$/, '').trim();
    }

    // Single raw Overpass query -> elements[] (or null on failure). Logged
    // and timed-out identically regardless of which place-type filter is
    // used by the caller.
    function queryOverpassRaw(query) {
        return new Promise((resolve) => {
            let url = `https://overpass-api.de/api/interpreter?data=${encodeURIComponent(query)}`;
            console.log('[TM-Obfuscate] Overpass query:', query);
            console.log('[TM-Obfuscate] GET', url);
            GM.xmlHttpRequest({
                method: 'GET',
                url,
                timeout: 30000, // must exceed the [timeout:25] set in the query itself, plus margin
                headers: { 'Accept': '*/*', 'User-Agent': 'TimoCom-Cargo-Formatter-UserScript' },
                onload: (resp) => {
                    try {
                        console.log('[TM-Obfuscate] overpass response status=%s bodyLength=%s', resp.status, (resp.responseText || '').length);
                        if (resp.status !== 200) { console.warn('[TM-Obfuscate] overpass non-200 body:', (resp.responseText || '').slice(0, 500)); resolve(null); return; }
                        let data = JSON.parse(resp.responseText);
                        let elements = data.elements || [];
                        console.log('[TM-Obfuscate] overpass raw element count:', elements.length);
                        resolve(elements);
                    } catch (e) { console.error('[TM-Obfuscate] overpass response parse error:', e); resolve(null); }
                },
                onerror: (err) => { console.error('[TM-Obfuscate] overpass request ERROR (likely blocked/CORS/permission):', err); resolve(null); },
                ontimeout: () => { console.warn('[TM-Obfuscate] overpass request TIMED OUT'); resolve(null); }
            });
        });
    }

    function elementToCity(el) {
        let pop = parseInt(String((el.tags && el.tags.population) || '0').replace(/[^\d]/g, ''), 10);
        return { name: (el.tags && el.tags.name) || '', lat: el.lat, lon: el.lon, population: isNaN(pop) ? 0 : pop };
    }

    function isUsableCity(c) {
        return !!(c.name && c.population > 0 && typeof c.lat === 'number' && typeof c.lon === 'number' && !isNaN(c.lat) && !isNaN(c.lon));
    }

    // Two-tier query: city/town first — a much smaller dataset than
    // including villages, which matters in densely-mapped regions (e.g. the
    // Rhine-Ruhr area around Cologne) where a village-inclusive query over a
    // 100km+ radius can return well over a thousand nodes and risks a slow
    // response or timeout even with the raised client timeout above. Only
    // broadens to villages if the city/town result alone is thin — keeping
    // sparse-area coverage without paying the cost everywhere else.
    async function fetchCitiesByPopulation(lat, lon, minKm, maxKm, excludeName, rankingMode, maxResults = 50) {
        if (isNaN(lat) || isNaN(lon)) {
            console.error('[TM-Obfuscate] fetchCitiesByPopulation called with a non-numeric origin (lat=%s, lon=%s) — refusing to query Overpass with garbage coordinates', lat, lon);
            return [];
        }
        let radiusM = Math.round(maxKm * 1000);
        let effectiveMinKm = Math.max(minKm, 2);

        let cityTownQuery = `[out:json][timeout:25];node(around:${radiusM},${lat},${lon})["place"~"^(city|town)$"]["population"];out body;`;
        let elements = await queryOverpassRaw(cityTownQuery);
        let rawCities = (elements || []).map(elementToCity).filter(isUsableCity);

        let inRangeCount = rawCities.filter(c => {
            let d = haversineKm(lat, lon, c.lat, c.lon);
            return d >= effectiveMinKm && d <= maxKm;
        }).length;

        if (inRangeCount < 5) {
            console.log('[TM-Obfuscate] city/town-only query found only %s in range — broadening to include villages', inRangeCount);
            let villageQuery = `[out:json][timeout:25];node(around:${radiusM},${lat},${lon})["place"="village"]["population"];out body;`;
            let villageElements = await queryOverpassRaw(villageQuery);
            if (villageElements) {
                rawCities = rawCities.concat(villageElements.map(elementToCity).filter(isUsableCity));
            }
        }

        // Hard safety floor: never treat anything closer than 2km as a
        // valid "nearby" pick, regardless of how low minKm is configured.
        // This exists because the postcode's geocoded centroid and OSM's
        // node for that SAME city are rarely at the exact same point —
        // without this, the real origin city can slip through a low/zero
        // minKm and get picked as an "obfuscated" result, defeating the
        // whole point.
        let cities = rawCities
            .map(c => ({ ...c, distanceKm: haversineKm(lat, lon, c.lat, c.lon) }))
            .filter(c => c.distanceKm >= effectiveMinKm && c.distanceKm <= maxKm);

        // Belt-and-suspenders: also exclude an EXACT name match with the
        // origin's own place, in case it's just far enough away (by the
        // centroid/node mismatch the 2km floor above guards against) to
        // pass the distance filter anyway. Deliberately NOT a substring
        // match — that previously excluded genuinely distinct towns that
        // merely share a name fragment with the origin (e.g. "Homburg" is a
        // real, different city from "Bad Homburg"), which could wipe out
        // the whole candidate list and force a fallback for the wrong
        // reason.
        if (excludeName) {
            let target = normalizeNameForCompare(excludeName);
            if (target) {
                cities = cities.filter(c => normalizeNameForCompare(c.name) !== target);
            }
        }

        // OSM sometimes has more than one node for the same place (e.g. a
        // city plus a separately-tagged district sharing its name) — keep
        // only the highest-population entry per name so the same place
        // can't occupy two different ranks.
        let byName = new Map();
        for (let c of cities) {
            let k = c.name.toLowerCase();
            if (!byName.has(k) || byName.get(k).population < c.population) byName.set(k, c);
        }
        cities = Array.from(byName.values());

        // Score by the selected ranking mode. "population" (default)
        // preserves the original largest-first behavior; the other two fold
        // proximity in so a much-closer mid-sized town can outrank a huge
        // city sitting near the edge of the radius.
        let scoreFn;
        if (rankingMode === 'pop_per_dist') {
            scoreFn = c => c.population / Math.max(c.distanceKm, 0.1);
        } else if (rankingMode === 'pop_per_dist2') {
            scoreFn = c => c.population / Math.max(c.distanceKm * c.distanceKm, 0.01);
        } else {
            scoreFn = c => c.population;
        }
        cities.sort((a, b) => scoreFn(b) - scoreFn(a));
        console.log('[TM-Obfuscate] after distance+name filtering: %s candidates remain, ranked by %s (top 5: %j)', cities.length, rankingMode || 'population', cities.slice(0, 5).map(c => `${c.name}(pop=${c.population}, dist=${c.distanceKm.toFixed(1)}km, score=${scoreFn(c).toFixed(1)})`));

        return cities.slice(0, maxResults);
    }

    // --- Per-Origin City Walk State ---
    // Keyed by the real postcode being obfuscated, so loading vs. delivery —
    // and different offers — each walk their own independently-fetched,
    // population-sorted city list. `nextIndex` is the city this origin will
    // hand out NEXT; once the list is exhausted it wraps back to the largest.
    let cityWalkState = {};

    // Resolves a single Overpass city node to a full "[Country Code]
    // [postcode] [city]" place via Nominatim (reusing the existing
    // rate-limit-safe, retrying reverseGeocode). Falls back to the Overpass
    // node's own name if Nominatim's reverse lookup doesn't return one.
    async function resolveCityPlace(city) {
        let place = await reverseGeocode(city.lat, city.lon);
        if (!place) return null;

        // zoom=14 (suburb/quarter level) sometimes lands on a feature with
        // no postcode tag at all in OSM. A closer, building-level lookup at
        // the exact same coordinate occasionally finds one even when the
        // broader area-level query didn't — worth one extra (throttled)
        // call rather than just shipping an address with no postcode.
        if (!place.postcode) {
            console.log('[TM-Obfuscate] no postcode at zoom=14 for %s — retrying at zoom=18', city.name);
            let finer = await reverseGeocode(city.lat, city.lon, 18);
            if (finer && finer.postcode) place.postcode = finer.postcode;
        }

        // Always use the Overpass-ranked city's own name, not whatever
        // Nominatim's reverse lookup resolves at that exact coordinate — at
        // city-level precision that can land on a suburb, district, or other
        // nearby named feature instead of the place we actually picked,
        // which made the walk look out of order even when it wasn't.
        // Nominatim is only trusted here for country code + postcode.
        place.city = city.name;
        return place;
    }

    // Kicks off resolving whatever city is up next, in the background,
    // without blocking the caller — this is the "look for next city AFTER
    // the content is copied" step. A no-op if a prefetch is already done or
    // already in flight, so it's safe to call defensively.
    function schedulePrefetch(state) {
        if (!state || !state.cities.length) return;
        if (state.prefetched || state.prefetchInFlight) return;
        let idx = state.nextIndex;
        let city = state.cities[idx % state.cities.length];
        state.prefetchInFlight = resolveCityPlace(city)
            .then(place => {
                state.prefetched = { index: idx, place };
                state.prefetchInFlight = null;
                return place;
            })
            .catch(() => { state.prefetchInFlight = null; });
    }

    // Called once per copy (after the clipboard write) for each obfuscated
    // end, so the NEXT city is already resolving while the user reads/pastes
    // what they just copied — by the time they click again, it's usually
    // already there waiting, with zero perceived wait.
    function ensureObfuscationPrefetch(stop) {
        if (!stop) return;
        let cc = (stop.country || '').trim();
        let parts = splitPostcodeAndCity(stop.city);
        let geoLookupKey = parts.postcode ? postcodeForGeocode(parts.postcode) : parts.cityName;
        if (!cc || !geoLookupKey) return;
        let state = cityWalkState[`${cc.toUpperCase()}:${geoLookupKey}`];
        schedulePrefetch(state);
    }

    async function getObfuscatedNearbyPlace(stop, maxResults = 50) {
        try {
            let oc = config.posting.locationObfuscation;
            let cc = (stop.country || '').trim();
            let originParts = splitPostcodeAndCity(stop.city);
            let hasPostcode = !!originParts.postcode;
            let geoLookupKey = hasPostcode ? postcodeForGeocode(originParts.postcode) : originParts.cityName;
            console.log('[TM-Obfuscate] resolving stop: country=%s rawCity=%j -> hasPostcode=%s geoLookupKey=%j', cc, stop.city, hasPostcode, geoLookupKey);
            if (!cc || !geoLookupKey) { console.warn('[TM-Obfuscate] missing country or geoLookupKey — cannot proceed, falling back'); return null; }

            let key = `${cc.toUpperCase()}:${geoLookupKey}`;
            let state = cityWalkState[key];

            if (!state) {
                // Initial lookup for this real address: nothing cached yet,
                // so this is the one case allowed to resolve before copying.
                // Addresses with an actual postcode go through Zippopotam;
                // addresses with only a city name (no postcode at all) go
                // through Nominatim's forward search instead — Zippopotam
                // has no concept of looking something up by name.
                console.log('[TM-Obfuscate] no cached city list for key=%s — doing initial lookup via %s', key, hasPostcode ? 'Zippopotam (postcode)' : 'Nominatim search (city name)');
                let origin = hasPostcode
                    ? await geocodePostcode(cc, geoLookupKey)
                    : await geocodeCityName(cc, cleanCityNameForSearch(geoLookupKey));

                // Zippopotam doesn't recognize every postcode it's handed —
                // e.g. Portugal's real format is "NNNN-NNN" and TimoCom may
                // only show the truncated "NNNN" part, which 404s; the UK's
                // alphanumeric postcodes aren't in Zippopotam's coverage at
                // all. Rather than give up, fall back to a name-based search
                // if we have a city name to try.
                if (!origin && hasPostcode && originParts.cityName) {
                    let cleanName = cleanCityNameForSearch(originParts.cityName);
                    console.log('[TM-Obfuscate] postcode lookup failed — retrying by city name "%s" instead', cleanName);
                    origin = await geocodeCityName(cc, cleanName);
                }

                console.log('[TM-Obfuscate] origin coordinates:', origin);
                if (!origin) { console.warn('[TM-Obfuscate] FAILED to geocode the origin address itself — falling back to real address. Check the network tab for the zippopotam/nominatim request.'); return null; }

                let minKm = Math.min(oc.minKm, oc.maxKm);
                let maxKm = Math.max(oc.minKm, oc.maxKm);
                let cities = await fetchCitiesByPopulation(origin.lat, origin.lon, minKm, maxKm, cleanCityNameForSearch(originParts.cityName), oc.rankingMode, maxResults);
                console.log('[TM-Obfuscate] candidate cities found within %s-%skm: %s', minKm, maxKm, cities.length, cities.slice(0, 10));
                if (!cities.length) { console.warn('[TM-Obfuscate] Overpass returned ZERO usable candidate cities — falling back to real address. Either the area genuinely has no OSM population data in range, or the Overpass request itself failed (check Network tab for overpass-api.de).'); return null; }

                state = { cities, nextIndex: 0, prefetched: null, prefetchInFlight: null };
                cityWalkState[key] = state;

                let place = await resolveCityPlace(cities[0]);
                console.log('[TM-Obfuscate] resolved first (largest) city ->', place);
                state.nextIndex = 1 % cities.length;
                return place;
            }

            if (!state.cities.length) { console.warn('[TM-Obfuscate] cached state has an empty city list for key=%s — falling back', key); return null; }

            // Largest-first walk: use whatever was already prefetched after
            // the previous copy (the common, instant case); if the user
            // clicked faster than the prefetch finished, fall back to
            // resolving right now instead of returning nothing.
            if (state.prefetched && state.prefetched.index === state.nextIndex) {
                let place = state.prefetched.place;
                console.log('[TM-Obfuscate] using prefetched city (instant) ->', place);
                state.prefetched = null;
                state.nextIndex = (state.nextIndex + 1) % state.cities.length;
                return place;
            }
            if (state.prefetchInFlight) {
                console.log('[TM-Obfuscate] prefetch was still in flight — awaiting it now');
                let place = await state.prefetchInFlight;
                console.log('[TM-Obfuscate] awaited in-flight prefetch ->', place);
                state.prefetched = null;
                state.prefetchInFlight = null;
                state.nextIndex = (state.nextIndex + 1) % state.cities.length;
                return place;
            }

            console.log('[TM-Obfuscate] no prefetch ready at all — resolving city at index %s right now', state.nextIndex);
            let city = state.cities[state.nextIndex % state.cities.length];
            let place = await resolveCityPlace(city);
            console.log('[TM-Obfuscate] resolved ->', place);
            state.nextIndex = (state.nextIndex + 1) % state.cities.length;
            return place;
        } catch (e) {
            console.error('[TM-Obfuscate] EXCEPTION in getObfuscatedNearbyPlace — this was previously silently swallowed:', e);
            return null;
        }
    }

    // --- Posting Export: Loading/Delivery Address Builder ---
    // Produces "[Country Code] [postcode] [city]" for the loading stop and
    // (when there's a genuinely different stop) the delivery stop. Each end's
    // obfuscation lookup runs concurrently; a single-stop load simply has no
    // delivery address (the line gets omitted).
    async function buildPostingAddresses(stops) {
        let oc = config.posting.locationObfuscation;
        let loadingStop = stops.length ? stops[0] : null;
        let deliveryStop = stops.length > 1 ? stops[stops.length - 1] : null;

        let loadingLookup = (oc.obfuscateLoading && loadingStop)
            ? getObfuscatedNearbyPlace(loadingStop, 5)
            : Promise.resolve(null);
        let deliveryLookup = (oc.obfuscateDelivery && deliveryStop)
            ? getObfuscatedNearbyPlace(deliveryStop, 1)
            : Promise.resolve(null);

        let [altLoadingPlace, altDeliveryPlace] = await Promise.all([loadingLookup, deliveryLookup]);
        if (oc.obfuscateLoading) console.log('[TM-Obfuscate] loading result: %s', altLoadingPlace ? 'OBFUSCATED -> ' + JSON.stringify(altLoadingPlace) : 'FELL BACK to real address');
        if (oc.obfuscateDelivery) console.log('[TM-Obfuscate] delivery result: %s', altDeliveryPlace ? 'OBFUSCATED -> ' + JSON.stringify(altDeliveryPlace) : 'FELL BACK to real address');

        let loadingAddress = altLoadingPlace ? buildAddressLineFromPlace(altLoadingPlace) : buildAddressLine(loadingStop);
        let deliveryAddress = altDeliveryPlace ? buildAddressLineFromPlace(altDeliveryPlace) : buildAddressLine(deliveryStop);

        return { loadingAddress, deliveryAddress };
    }

    // --- Template Compiler Engine ---
    function generateFormattedTemplate(targetUrl) {
        let stops = collectStops();
        let first = stops.length ? stops[0] : {};
        let last = stops.length ? stops[stops.length - 1] : {};

        // First loading stop -> "Loading"; last stop -> "Delivery"
        let dateRangeFirst = mergeRange(first.eDate, first.lDate, "27.05");
        let dateRangeLast = mergeRange(last.eDate, last.lDate, "28.05");

        let rawTimeFirst = first.lTime || first.eTime || "12:00";
        let rawTimeLast = last.lTime || last.eTime || "13:00";

        let finalTimeFirst = offsetTimeStr(rawTimeFirst, config.offsetLoad);
        let finalTimeLast = offsetTimeStr(rawTimeLast, config.offsetDel);

        let ldm = extractDigits(getTextByTestId('FreightDescriptionView/freightLength')) || "3";
        let wt = extractDigits(getTextByTestId('FreightDescriptionView/freightWeight')) || "0.65";

        let desc = getTextByTestId('FreightDescriptionView/typeOfGoods');
        let addInfo = getTextByTestId('FreightDescriptionView/additionalInformation');
        let remarks = getTextByTestId('RemarksView/external-remarks');
        let combinedText = `${desc} ${addInfo} ${remarks}`;

        let lines = [];

        if (config.prefixText) lines.push(config.prefixText);
        if (config.showRoute) lines.push(`Route: ${targetUrl || "Generating map route..."}`);
        if (config.showLoading) lines.push(`Loading: ${dateRangeFirst} ${finalTimeFirst}`);
        if (config.showDelivery) lines.push(`Delivery: ${dateRangeLast} ${finalTimeLast}`);
        if (config.showDims) lines.push(`LDM/ Wt: ${ldm}m ${wt}t`);

        if (config.showNote && /lift|adr|pallet|plt|pal|epal/i.test(combinedText)) {
            let noteContent = desc || addInfo || remarks || "LIFT/ADR/Pallets";
            lines.push(`Note: ${noteContent}`);
        }

        if (config.suffixText) lines.push(config.suffixText);

        return lines.join('\n');
    }

    // --- Posting Export: Template Compiler ---
    // Independent from generateFormattedTemplate above: own field toggles, own
    // hour offsets, randomized LDM/weight, keyword-only notes, and the two
    // static tag lines (Typ nadwozia / Wielkość pojazdu).
    // --- Posting Export: Shared Line Builder ---
    // All the field-toggle / label / jitter / notes / tag logic lives here,
    // fed by a plain data object — so the live-DOM flow and the "paste
    // freight info" flow produce byte-identical output from the same inputs.
    function buildPostingLines(data) {
        let pc = config.posting;

        let finalTimeFirst = offsetTimeStr(data.timeFirst, pc.offsetLoad);
        let finalTimeLast = offsetTimeStr(data.timeLast, pc.offsetDel);
        let dims = jitterDims(data.ldmRaw, data.wtRaw);
        let notesStr = extractNotesKeywords(data.notesText);

        let lines = [];

        if (pc.showLoadingAddress && data.loadingAddress) lines.push(withLabel(pc.labels.loadingAddress, data.loadingAddress));
        if (pc.showDeliveryAddress && data.deliveryAddress) lines.push(withLabel(pc.labels.deliveryAddress, data.deliveryAddress));

        let loadingVal = `${slashDate(data.dateRangeFirst)} ${finalTimeFirst}`.trim();
        if (pc.showLoading && loadingVal) lines.push(withLabel(pc.labels.loading, loadingVal));

        let deliveryVal = `${slashDate(data.dateRangeLast)} ${finalTimeLast}`.trim();
        if (pc.showDelivery && deliveryVal) lines.push(withLabel(pc.labels.delivery, deliveryVal));

        if (pc.showDims) lines.push(withLabel(pc.labels.dims, `${dims.ldm}m ${dims.wt}t`));
        if (pc.showNotes && notesStr) lines.push(withLabel(pc.labels.notes, notesStr));

        if (pc.showBodyType) {
            let sel = (pc.bodyTypeTags || []).filter(t => t.enabled).map(t => t.text);
            if (sel.length) lines.push(withLabel(pc.labels.bodyType, sel.join('; ')));
        }
        if (pc.showVehicleSize) {
            let sel = (pc.vehicleSizeTags || []).filter(t => t.enabled).map(t => t.text);
            if (sel.length) lines.push(withLabel(pc.labels.vehicleSize, sel.join('; ')));
        }

        return lines.join('\n');
    }

    // --- Posting Export: Template Compiler (live DOM source) ---
    function generatePostingTemplate(addresses) {
        let stops = collectStops();
        let first = stops.length ? stops[0] : {};
        let last = stops.length ? stops[stops.length - 1] : {};

        let ldmRaw = extractDigits(getTextByTestId('FreightDescriptionView/freightLength')) || "3";

        let desc = getTextByTestId('FreightDescriptionView/typeOfGoods');
        let addInfo = getTextByTestId('FreightDescriptionView/additionalInformation');
        let remarks = getTextByTestId('RemarksView/external-remarks');

        return buildPostingLines({
            loadingAddress: addresses.loadingAddress,
            deliveryAddress: addresses.deliveryAddress,
            dateRangeFirst: mergeRange(first.eDate, first.lDate, "27.05"),
            dateRangeLast: mergeRange(last.eDate, last.lDate, "28.05"),
            timeFirst: first.lTime || first.eTime || "12:00",
            timeLast: last.lTime || last.eTime || "13:00",
            ldmRaw,
            wtRaw,
            notesText: `${desc} ${addInfo} ${remarks}`
        });
    }

    // --- Posting Export: Template Compiler (pasted-text source) ---

        });
    }

    async function executePostingCopy() {
        if (!currentOfferId) return;

        const postBtn = shadowRoot.getElementById('tm-cargo-post-btn');
        const oc = config.posting.locationObfuscation;
        const obfuscating = oc.obfuscateLoading || oc.obfuscateDelivery;
        console.log('[TM-Obfuscate] click. obfuscateLoading=%s obfuscateDelivery=%s minKm=%s maxKm=%s', oc.obfuscateLoading, oc.obfuscateDelivery, oc.minKm, oc.maxKm);

        try {
            const stops = collectStops();
            console.log('[TM-Obfuscate] stops:', JSON.parse(JSON.stringify(stops)));

            if (postBtn && obfuscating) {
                postBtn.innerText = '⏳ Finding nearby city...';
                postBtn.style.pointerEvents = 'none';
            }

            const addresses = await buildPostingAddresses(stops);
            console.log('[TM-Obfuscate] final addresses used:', addresses);
            const finalOutput = generatePostingTemplate(addresses);
            GM_setClipboard(finalOutput, "text");

            // Now that this round's pick is locked in and copied, start
            // resolving the NEXT city in the background for whichever ends
            // are obfuscated — ready before the user clicks again.
            if (oc.obfuscateLoading) ensureObfuscationPrefetch(stops[0]);
            if (oc.obfuscateDelivery && stops.length > 1) ensureObfuscationPrefetch(stops[stops.length - 1]);

            if (!postBtn) return;

            postBtn.innerText = '✅ Copied!';

            postBtn.style.pointerEvents = 'auto';
            setTimeout(() => {
                postBtn.innerText = POST_BTN_DEFAULT_TEXT;
                postBtn.style.backgroundColor = baseBg;
            }, 1800);

        } catch (err) {
            console.error("Posting clipboard failure:", err);
            if (postBtn) {
                postBtn.innerText = POST_BTN_DEFAULT_TEXT;
                postBtn.style.pointerEvents = 'auto';
            }
        }
    }

    // --- Posting Settings Panel: Editable Tag List (Typ nadwozia / Wielkość pojazdu) ---
    // Built with plain DOM nodes (not innerHTML strings) so tag text never needs
    // HTML-escaping and rows can be added/removed live without re-rendering the
    // whole list (which would lose in-progress edits in other rows).
    function buildTagRow(tag) {
        const row = document.createElement('div');
        row.style.display = 'flex';
        row.style.alignItems = 'center';
        row.style.gap = '6px';

        const cb = document.createElement('input');
        cb.type = 'checkbox';
        cb.checked = tag.enabled !== false;
        cb.dataset.role = 'enabled';
        row.appendChild(cb);

        const txt = document.createElement('input');
        txt.type = 'text';
        txt.value = tag.text || '';
        txt.dataset.role = 'text';
        txt.style.flex = '1';
        txt.style.minWidth = '0';
        txt.style.fontSize = '12px';
        txt.style.boxSizing = 'border-box';
        txt.style.padding = '2px 4px';
        row.appendChild(txt);

        const rm = document.createElement('span');
        rm.innerHTML = '✖';
        rm.title = 'Remove tag';
        rm.style.cursor = 'pointer';
        rm.style.color = '#dc2626';
        rm.style.fontWeight = 'bold';
        rm.style.padding = '0 4px';
        rm.addEventListener('click', () => row.remove());
        row.appendChild(rm);

        return row;
    }

    function renderTagList(containerEl, tags) {
        containerEl.innerHTML = '';
        (tags || []).forEach(tag => containerEl.appendChild(buildTagRow(tag)));
    }

    // Rows with blank text are skipped (clearing the text field is the same as
    // deleting the row, in case someone forgets to hit the ✖).
    function readTagList(containerEl) {
        let tags = [];
        Array.from(containerEl.children).forEach(row => {
            const txt = row.querySelector('input[data-role="text"]');
            const cb = row.querySelector('input[data-role="enabled"]');
            const text = txt ? txt.value.trim() : '';
            if (text) tags.push({ text: text, enabled: cb ? cb.checked : true });
        });
        return tags;
    }

        });

        mapsBtn.style.borderRadius = '6px';
        mapsBtn.style.fontWeight = 'bold';
        mapsBtn.style.border = 'none';
        mapsBtn.style.pointerEvents = 'none';
        mapsBtn.style.cursor = 'not-allowed';
        mapsBtn.style.transition = 'all 0.2s ease';
        mapsBtn.style.boxSizing = 'border-box';
        mapsBtn.style.fontFamily = 'inherit';
        content.appendChild(mapsBtn);

        mapsBtn.addEventListener('click', () => {
            if (currentMapUrl) {
                window.open(currentMapUrl, '_blank', 'noopener,noreferrer');
            }
        });

        menu.appendChild(content);
        shadowRoot.appendChild(menu);

        // --- Event Listeners for Clickable Box and Hover Visuals ---
        statusText.addEventListener('click', () => {
            if (currentOfferId && currentShortUrl !== "Shortening link...") {
                executeUnifiedCopy();
            }
        });

        statusText.addEventListener('mouseenter', () => {
            if (currentOfferId && currentShortUrl !== "Shortening link...") {
                statusText.style.backgroundColor = '#f1f5f9';
                statusText.style.borderColor = '#94a3b8';
            }
        });

        statusText.addEventListener('mouseleave', () => {
            if (currentOfferId) {
                statusText.style.backgroundColor = '#f8fafc';
                statusText.style.borderColor = '#e2e8f0';
            }
        });

        actionBtn.addEventListener('click', executeUnifiedCopy);

        // --- Drag Mathematics Mechanics ---
        let isDragging = false, startX, startY, initialLeft, initialTop;
        header.addEventListener('mousedown', (e) => {
            isDragging = true;
            startX = e.clientX;
            startY = e.clientY;
            initialLeft = parseInt(menu.style.left || 0, 10);
            initialTop = parseInt(menu.style.top || 0, 10);
            e.preventDefault();
        });

        document.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            const deltaX = e.clientX - startX;
            const deltaY = e.clientY - startY;
            menu.style.left = (initialLeft + deltaX) + 'px';
            menu.style.top = (initialTop + deltaY) + 'px';
        });

        document.addEventListener('mouseup', () => {
            if (isDragging) {
                localStorage.setItem('tm-cargo-menu-left', menu.style.left);
                localStorage.setItem('tm-cargo-menu-top', menu.style.top);
            }
            isDragging = false;
        });
    }

    // --- State Synchronization Loop Engine ---
    async function processActiveRoute() {
        buildFloatingMenu();

        const statusText = shadowRoot.getElementById('tm-cargo-status-text');
        const actionBtn = shadowRoot.getElementById('tm-cargo-action-btn');
        const mapsBtn = shadowRoot.getElementById('tm-cargo-maps-btn');
        const codesBtn = shadowRoot.getElementById('tm-cargo-codes-btn');
        const postBtn = shadowRoot.getElementById('tm-cargo-post-btn');
        if (!statusText || !actionBtn) return;

        const testCountryNode = document.querySelector('[data-testid="LoadingPlacesView/loadingPlaceCountry_0"]');

        if (testCountryNode) {
            let stops = collectStops();

            // Signature spans every stop so the route refreshes when the
            // selected offer changes, including multi-stop / round trips.
            let uniqueSignature = stops.map(s => `${s.country}_${s.city}`).join('|');

            if (currentOfferId !== uniqueSignature) {
                currentOfferId = uniqueSignature;

                let longUrl = buildMapsUrl(stops);

                currentMapUrl = longUrl;

                // Full directions URL is ready immediately (no shortening required)
                if (mapsBtn && longUrl) {
                    mapsBtn.style.backgroundColor = '#0f9d58';
                    mapsBtn.style.color = '#ffffff';
                    mapsBtn.style.pointerEvents = 'auto';
                    mapsBtn.style.cursor = 'pointer';
                    mapsBtn.onmouseenter = () => mapsBtn.style.backgroundColor = '#0b8043';
                    mapsBtn.onmouseleave = () => mapsBtn.style.backgroundColor = '#0f9d58';
                }

                // Codes/postcodes are read straight from the DOM (no shortening)
                if (codesBtn) {
                    codesBtn.style.backgroundColor = '#7c3aed';
                    codesBtn.style.color = '#ffffff';
                    codesBtn.style.pointerEvents = 'auto';
                    codesBtn.style.cursor = 'pointer';
                    codesBtn.onmouseenter = () => codesBtn.style.backgroundColor = '#6d28d9';
                    codesBtn.onmouseleave = () => codesBtn.style.backgroundColor = '#7c3aed';
                }

                // Posting export is also read straight from the DOM (no shortening)
                if (postBtn) {
                    postBtn.style.backgroundColor = '#ea580c';
                    postBtn.style.color = '#ffffff';
                    postBtn.style.pointerEvents = 'auto';
                    postBtn.style.cursor = 'pointer';
                    postBtn.onmouseenter = () => postBtn.style.backgroundColor = '#c2410c';
                    postBtn.onmouseleave = () => postBtn.style.backgroundColor = '#ea580c';
                }

                currentShortUrl = "Shortening link...";
                statusText.innerText = generateFormattedTemplate(currentShortUrl);
                statusText.style.cursor = 'wait';
                statusText.title = 'Generating route links...';

                actionBtn.style.backgroundColor = '#f59e0b';
                actionBtn.innerText = '⏳ Shortening Link...';
                actionBtn.style.pointerEvents = 'none';

                let compressedUrl = await fetchShortenedUrl(longUrl);

                if (currentOfferId === uniqueSignature) {
                    currentShortUrl = compressedUrl;
                    statusText.innerText = generateFormattedTemplate(currentShortUrl);
                    statusText.style.cursor = 'pointer';
                    statusText.title = 'Click inside this text block to copy everything!';

                    actionBtn.innerText = '📋 Copy Short Route Template';
                    actionBtn.style.backgroundColor = '#0284c7';
                    actionBtn.style.color = '#ffffff';
                    actionBtn.style.pointerEvents = 'auto';
                    actionBtn.style.cursor = 'pointer';

                    actionBtn.onmouseenter = () => actionBtn.style.backgroundColor = '#0369a1';
                    actionBtn.onmouseleave = () => actionBtn.style.backgroundColor = '#0284c7';
                }
            }
        } else {
            if (currentOfferId !== "") {
                currentOfferId = "";
                currentShortUrl = "";
                currentMapUrl = "";
                statusText.innerHTML = '<em>Select a freight offer details panel...</em>';
                statusText.style.cursor = 'default';
                statusText.title = '';

                actionBtn.innerText = '📋 Copy Short Route Template';
                actionBtn.style.backgroundColor = '#e2e8f0';
                actionBtn.style.color = '#94a3b8';
                actionBtn.style.pointerEvents = 'none';
                actionBtn.style.cursor = 'not-allowed';
                actionBtn.onmouseenter = null;
                actionBtn.onmouseleave = null;

                if (mapsBtn) {
                    mapsBtn.style.backgroundColor = '#e2e8f0';
                    mapsBtn.style.color = '#94a3b8';
                    mapsBtn.style.pointerEvents = 'none';
                    mapsBtn.style.cursor = 'not-allowed';
                    mapsBtn.onmouseenter = null;
                    mapsBtn.onmouseleave = null;
                }

                if (codesBtn) {
                    codesBtn.style.backgroundColor = '#e2e8f0';
                    codesBtn.style.color = '#94a3b8';
                    codesBtn.style.pointerEvents = 'none';
                    codesBtn.style.cursor = 'not-allowed';
                    codesBtn.onmouseenter = null;
                    codesBtn.onmouseleave = null;
                }

                if (postBtn) {
                    postBtn.style.backgroundColor = '#e2e8f0';
                    postBtn.style.color = '#94a3b8';
                    postBtn.style.pointerEvents = 'none';
                    postBtn.style.cursor = 'not-allowed';
                    postBtn.onmouseenter = null;
                    postBtn.onmouseleave = null;
                }
            }
        }
    }

    const observer = new MutationObserver(processActiveRoute);
    observer.observe(document.body, { childList: true, subtree: true });
    setInterval(processActiveRoute, 350);
})();
